\section{Motivación y Desafío del Retail}
% ==============================================================================

\begin{frame}{El Problema Logístico en Retail de Perecederos}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{El Dilema del Inventario}}
        \begin{itemize}
          \item \textbf{Asimetría de Costes:} Perder una venta por rotura ($c_u = 4{,}0$) es \textbf{4 veces más penalizado} que desechar exceso de stock fresco ($c_o = 1{,}0$).
          \item \textbf{Censura de Demanda por Rotura:} Cuando el stock se agota ($I_t = 0$), las ventas observadas solo registran lo disponible, \textbf{no la demanda real latente}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{La Falla de los Modelos Clásicos}}
        \begin{itemize}
          \item \textbf{Trampa del MAE/MSE:} Entrenar sobre demanda censurada aprende a subestimar el volumen en días de alta rotura.
          \item \textbf{Incertidumbre no calibrada:} Previsiones puntuales sin intervalos con garantía estadística fiable.
          \item \textbf{Obsolescencia en producción:} Los modelos estáticos degradan su nivel de servicio rápidamente.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Imputación Supervisada de Demanda Latente}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{\textbf{Formulación Matemática de la Censura}}
        $$Y_t^{\text{obs}} = \begin{cases} D_t & \text{si } I_t > 0 \text{ (no rotura)} \\ \min(D_t, C_t) & \text{si } I_t = 0 \text{ (rotura de stock)} \end{cases}$$
        \begin{itemize}
          \item La demanda real $D_t$ queda censurada por la derecha cuando $I_t = 0$.
          \item \textbf{Imputador Supervisado:} Regresor LightGBM entrenado sobre días no censurados con censura sintética.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\textbf{Impacto en la Calidad del Dato}}
        \begin{itemize}
          \item Reconstruye el pico de demanda oculto tras las roturas.
          \item Elimina el sesgo de subestimación sistemático.
          \item Sintonizado mediante Optuna sobre $500$ series ($100$ épocas).
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% ==============================================================================
\begin{frame}{Ingeniería de Características y Contrato Causal}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{43 Variables Predictivas Causales}}
        \begin{itemize}
          \item \textbf{Lags de Demanda:} $t-1, t-2, t-7, t-14, t-21$.
          \item \textbf{Estadísticos Móviles:} Medias, desviaciones estándar, mínimos y máximos en ventanas de $7, 14, 28$ días.
          \item \textbf{Precios y Promociones:} Descuento porcentual, ratio de precio relativo vs histórico.
          \item \textbf{Calendarios:} Día de la semana, día del mes, fin de semana.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Garantía Anti-Leakage}}
        \begin{itemize}
          \item Transformaciones calculadas \textbf{estrictamente en el pasado} ($t' < t$).
          \item Validación temporal mediante $K=3$ folds walk-forward sin solapamiento.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ==============================================================================
\begin{frame}{Calibración Conformal (Mondrian Conformal Prediction)}
  \begin{block}{\textbf{Intervalos con Garantías Teóricas Finitas}}
    Buscamos un intervalo $C(X_{n+1}) = [\hat{q}_{\alpha/2}(X_{n+1}) - \hat{s}, \hat{q}_{1-\alpha/2}(X_{n+1}) + \hat{s}]$ tal que:
    $$P(Y_{n+1} \in C(X_{n+1})) \ge 1 - \alpha$$
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\textbf{Ventajas de Mondrian CP}}
        \begin{itemize}
          \item \textbf{No paramétrico:} Sin asunción gaussiana de residuos.
          \item \textbf{Estratificación por Categoría:} Calibra factores $\hat{s}_g$ independientes por familia de producto.
          \item Válido en muestras finitas bajo intercambiabilidad.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Factor de Corrección $\hat{s}$}}
        $$\hat{s} = \text{Quantile}\left( \frac{\lceil (n+1)(1-\alpha) \rceil}{n}, \{R_i\}_{i=1}^n \right)$$
        donde $R_i$ son los residuos no conformistas del conjunto de calibración.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ==============================================================================
\begin{frame}{Toma de Decisión: Optimización Newsvendor Asimétrica}
  \begin{block}{\textbf{Formulación Financiera Newsvendor}}
    La cantidad óptima de pedido $q^*$ minimiza la pérdida esperada asimétrica:
    $$L(q, D) = c_o \max(0, q - D) + c_u \max(0, D - q)$$
  \end{block}

  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\textbf{Fractil Crítico $\tau$}}
        $$\tau = \frac{c_u}{c_u + c_o} = \frac{4{,}0}{4{,}0 + 1{,}0} = 0{,}80$$
        \begin{itemize}
          \item $c_u = 4{,}0$ u.m. (coste de rotura de stock).
          \item $c_o = 1{,}0$ u.m. (coste de sobrestock perecedero).
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Cálculo de Pedido $q^*$}}
        $$q^* = \hat{F}_D^{-1}(\tau) = \hat{y}_{\text{pred}} + \hat{s}_{\text{conformal}}(\tau)$$
        Alinea la predicción probabilística directamente con el beneficio financiero del negocio.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
